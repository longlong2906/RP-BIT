cmake_minimum_required(VERSION 3.20)
project(bitstar_25d VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(TERRAIN_WITH_OMPL "Build the real OMPL BIT* worker" ON)
include(CTest)
add_library(terrain25d src/Terrain.cpp src/FlatDirectCertificate.cpp)
target_include_directories(terrain25d PUBLIC include)
if(MSVC)
  target_compile_options(terrain25d PRIVATE /W4)
else()
  target_compile_options(terrain25d PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(terrain_cli apps/terrain_cli.cpp)
target_link_libraries(terrain_cli PRIVATE terrain25d)
add_executable(terrain_tests tests/unit/terrain_tests.cpp)
target_link_libraries(terrain_tests PRIVATE terrain25d)
add_test(NAME terrain_unit COMMAND terrain_tests)
add_executable(traversability_tests tests/unit/traversability_tests.cpp)
target_link_libraries(traversability_tests PRIVATE terrain25d)
add_test(NAME traversability_unit COMMAND traversability_tests)
add_executable(flat_direct_certificate_tests tests/test_flat_direct_certificate.cpp)
target_link_libraries(flat_direct_certificate_tests PRIVATE terrain25d)
add_test(NAME flat_direct_certificate COMMAND flat_direct_certificate_tests)
set_tests_properties(flat_direct_certificate PROPERTIES FIXTURES_SETUP correctness_core)
add_executable(a3_preprocessing_tests tests/test_a3_preprocessing.cpp)
target_link_libraries(a3_preprocessing_tests PRIVATE terrain25d)
add_test(NAME a3_preprocessing COMMAND a3_preprocessing_tests)
set_tests_properties(a3_preprocessing PROPERTIES FIXTURES_SETUP correctness_core)
add_executable(a3_preprocessing_benchmark apps/a3_preprocessing_benchmark.cpp)
target_link_libraries(a3_preprocessing_benchmark PRIVATE terrain25d)
find_package(Python3 COMPONENTS Interpreter REQUIRED)
add_test(NAME independent_geometry COMMAND ${Python3_EXECUTABLE}
  ${CMAKE_SOURCE_DIR}/tests/test_geometry.py --cli $<TARGET_FILE:terrain_cli>)
set_tests_properties(terrain_unit independent_geometry PROPERTIES FIXTURES_SETUP correctness_core)
add_test(NAME traversability_independent_oracle COMMAND ${Python3_EXECUTABLE}
  ${CMAKE_SOURCE_DIR}/tests/test_traversability_oracle.py --cli $<TARGET_FILE:terrain_cli>)
add_test(NAME traversability_analysis COMMAND ${Python3_EXECUTABLE}
  ${CMAKE_SOURCE_DIR}/tests/test_traversability_analysis.py)
add_test(NAME traversability_protocol COMMAND ${Python3_EXECUTABLE}
  ${CMAKE_SOURCE_DIR}/tests/test_a3_traversability_protocol.py)
add_test(NAME a3_perf_protocol COMMAND ${Python3_EXECUTABLE}
  ${CMAKE_SOURCE_DIR}/tests/test_a3_traversability_perf_protocol.py)
set_tests_properties(a3_perf_protocol PROPERTIES FIXTURES_SETUP correctness_core)
set_tests_properties(traversability_unit traversability_independent_oracle traversability_analysis traversability_protocol
  PROPERTIES FIXTURES_SETUP correctness_core)
if(TERRAIN_WITH_OMPL)
  find_package(ompl 1.7.0 EXACT CONFIG REQUIRED)
  include(cmake/OmplTargets.cmake)
endif()
